1. Scope and session

Restated from doc 19 §1 and doc 18 §3.7. Every request carries a resolved principal and scope; the database
session carries the same facts as custom settings that the row-level security policies read.

// kernel.api.ScopeContext

public record ScopeContext(UUID userId, UUID deviceId, UUID homeEntityId,

                                     List<Scope> scopes, PolicyClass policyClass,

                                     Set<UUID> grantedEntities, Instant mfaAt, String lang, UUID correlationId) {

         public UUID entityId() { return activeScope().entityId(); }                // the scope chosen for this request

         public UUID locationId() { return activeScope().locationId(); } // null for entity-wide

}

// kernel.internal.ScopeFilter (web profile): token -> ScopeContext -> request attribute + MDC
// kernel.internal.ScopeConnectionCustomizer: at transaction start, on the same connection:
// SET LOCAL app.scope_entity_id = '<uuid>'; SET LOCAL app.scope_location_id = '<uuid or empty>';
// SET LOCAL app.scope_class = 'OWN' | 'PARTY' | 'FEDERATION_VIEW' | 'EXTERNAL_TIMEBOXED';
// SET LOCAL (not SET) because PgBouncer runs in transaction pooling: settings must die with the transaction.

Aspect                  Design

Active scope            A user with several scopes (Federation view, MPCS administrator of many shops) chooses one per request
                        through the X-Scope-Entity / X-Scope-Location headers; the filter verifies the choice is among the token's
                        scopes; a missing header with exactly one scope defaults to it; otherwise 400 scope.required.

Worker and ingest profiles No HTTP; the job runner and the batch ingestor construct a ScopeContext explicitly (system actor, entity from the
                                      job or the device) and call the same customizer. A transaction without a scope fails closed: the customizer sets
                                      app.scope_class = 'NONE' and RLS returns nothing.

-- immutability (doc 18 C-I1): a trigger raises on any UPDATE of an issued document except status

CREATE OR REPLACE FUNCTION kernel.document_immutable() RETURNS trigger LANGUAGE plpgsql AS $$

BEGIN

IF OLD.issued_at IS NOT NULL AND (to_jsonb(NEW) - 'status') IS DISTINCT FROM (to_jsonb(OLD) - 'status') THEN

    RAISE EXCEPTION 'document.immutable' USING ERRCODE = 'check_violation';

END IF; RETURN NEW;

END $$;

CREATE TRIGGER trg_document_immutable BEFORE UPDATE ON kernel.document FOR EACH ROW EXECUTE FUNCTION

    kernel.document_immutable();

GRANT INSERT ON kernel.document, kernel.document_line, kernel.document_link, kernel.document_state_history TO

    app_rw;
