    sku_id uuid PRIMARY KEY, sku_code varchar(12) NOT NULL UNIQUE, owner_entity_id uuid NOT NULL,

    status text NOT NULL DEFAULT 'DRAFT' CHECK (status IN ('DRAFT','LOCAL','SHARED','INACTIVE')),

    prior_status text CHECK (prior_status IN ('LOCAL','SHARED')),                  -- for reactivation

    short_name_en varchar(80) NOT NULL COLLATE en_icu, short_name_si varchar(80) COLLATE si_icu, short_name_ta

    varchar(80) COLLATE ta_icu,

    description_en text, description_si text, description_ta text,

    base_uom_code varchar(10) NOT NULL REFERENCES catalogue.uom,

    sold_by_weight boolean NOT NULL DEFAULT false, batch_tracked boolean NOT NULL DEFAULT false,

    expiry_tracked boolean NOT NULL DEFAULT false, has_printed_mrp boolean NOT NULL DEFAULT true,

    expiry_warning_days smallint, tax_category_id uuid NOT NULL,

    multi_mrp_policy text NOT NULL DEFAULT 'AUTO_LOWEST' CHECK (multi_mrp_policy IN

    ('AUTO_LOWEST','BARCODE_RESOLVED','PICKER')),

    origin_kind text NOT NULL DEFAULT 'PURCHASED' CHECK (origin_kind IN

    ('PURCHASED','REPACK_OUTPUT','LOCAL_SOURCED')),

    attributes jsonb NOT NULL DEFAULT '{}' CHECK (pg_column_size(attributes) <= 4096),

    CHECK (status <> 'SHARED' OR (short_name_si IS NOT NULL AND short_name_ta IS NOT NULL)), -- B-I1

    CHECK (NOT sold_by_weight OR base_uom_code = 'KG'),

    CHECK (NOT expiry_tracked OR batch_tracked)

6.1 Duplicate scoring

// steward/DuplicateScorer.score(candidate, existing) -> Score(value 0..1, reasons[])

key(s)   = nfc(s) -> lowercase -> strip punctuation -> transliterate(si|ta -> latin, table v1) -> tokens

brand(t) = first token if in brand-list (governed tag brand:* or a seeded brand file), else null

pack(t)  = tokens matching (\d+(\.\d+)?)(kg|g|l|ml|pkt|pcs) normalised to base unit

score    = 0.55 * jaroWinkler(key(candidate.nameEn), key(existing.nameEn))

         + 0.20 * [brand equal] + 0.15 * [pack equal within 2%] + 0.10 * [same base uom]

         + 0.30 if any barcode equal (capped at 1.0)                               // a shared barcode is near-certain

suspect  = score >= config catalogue.duplicate_threshold (0.85) and not previously DISMISSED for this pair

// applied on CreateSku, BulkImport (row vs catalogue and row vs row), and RegisterBatch (batch_no edit

distance <= 1

// for the same sku and supplier -> batch-level suspect flagged to the steward, batches still both registered)

6.2 Import

// steward/ImportService.run(sheet, mode, ctx)

rows = parse(sheet)                                            // xlsx or csv; columns: code?, name_en, name_si,

name_ta, base_uom, tax, flags, tags, barcodes[], case_factor, image_ref

for r in rows:

key = hash(r.rawCells); if importLog.seen(key) -> outcome DUPLICATE_ROW (idempotent re-upload)

errors = RowValidator.validate(r)                              // each error = {row, field, messageId, params}

suspects = scorer.against(catalogue).plus(scorer.withinSheet(rows))

if mode == VALIDATE: report.add(r, errors, suspects); continue

if errors.isEmpty(): tx { CreateSku(...); register barcodes; define CASE conversion; importLog.record(key) }

       ; report.accepted++

else report.add(r, errors)

return report                                                  // rendered by the import screen; downloadable as a

sheet
