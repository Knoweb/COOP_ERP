24A ­ M4 Trading Documents: Implementation
Guide

Everything the M4 developer needs: the trading schema as DDL on top of the kernel
document base, seed data including the posting map, package layout, the API slice,
every command as a handler specification with pseudocode for the four that carry the
module (GRN confirmation, allocation, settlement and its reversal, exposure), the
consumers, the snapshot contribution and the till-side GRN bundle contract for M6,
screens, tests and tickets. Self-contained.

Read this first. M4 is nine document types on one base. Build them in the order they occur in the world (order,
delivery, GRN, discrepancy, claim, invoice, payment) and make ConfirmGrnHandler the first thing that works end
to end with M2 and M5 stubs, because five other things hang off its event. Money is derived from links; if you find
yourself storing a balance that isn't recomputed by the same event that changes it, stop.

3. Schema

Restated from doc 24 §3 (CR-24-1). All extension tables key on document_id of the kernel base and are
partitioned alike by received_at (carried as a column for the partition key). Location
db/migration/m4trading/.

-- V0001__trading.sql (partition clauses and RLS lines abbreviated: every table follows 17A 6.3 plus the PARTY
        policy)

CREATE TABLE trading.doc_order (document_id uuid NOT NULL, received_at timestamptz NOT NULL, relationship_id
        uuid NOT NULL,

    buyer_entity_id uuid NOT NULL, seller_entity_id uuid NOT NULL, requested_eta date, committed_eta date,
           lock_at timestamptz,

    lock_override_by uuid, allocation_run_id uuid, version smallint NOT NULL DEFAULT 1, owner_entity_id uuid NOT
           NULL,

    PRIMARY KEY (received_at, document_id)) PARTITION BY RANGE (received_at);
CREATE TABLE trading.doc_order_line (line_id uuid NOT NULL, received_at timestamptz NOT NULL, document_id uuid

        NOT NULL,
    requested_qty numeric(14,3) NOT NULL CHECK (requested_qty > 0), allocated_qty numeric(14,3) NOT NULL DEFAULT

           0,
    fulfilled_qty numeric(14,3) NOT NULL DEFAULT 0, cancelled_qty numeric(14,3) NOT NULL DEFAULT 0,
    tier_price numeric(14,4), adjustment_type varchar(24), adjustment_value numeric(14,4), owner_entity_id uuid

           NOT NULL,
    CHECK (allocated_qty + cancelled_qty <= requested_qty), CHECK (fulfilled_qty <= allocated_qty),
    PRIMARY KEY (received_at, line_id)) PARTITION BY RANGE (received_at);
CREATE TABLE trading.allocation_run (run_id uuid PRIMARY KEY, seller_entity_id uuid NOT NULL, ran_at

        timestamptz NOT NULL,
    rule text NOT NULL, availability_snapshot jsonb NOT NULL, order_ids uuid[] NOT NULL, overrides jsonb NOT

           NULL DEFAULT '[]',
    ran_by uuid NOT NULL, owner_entity_id uuid NOT NULL);

CREATE TABLE trading.doc_delivery (document_id uuid NOT NULL, received_at timestamptz NOT NULL,
        seller_entity_id uuid NOT NULL,

    vehicle_ref text, driver_user_id uuid, driver_name text, dispatched_at timestamptz, route_ref text,
           owner_entity_id uuid NOT NULL,

    PRIMARY KEY (received_at, document_id)) PARTITION BY RANGE (received_at);
CREATE TABLE trading.doc_delivery_drop (drop_id uuid NOT NULL, received_at timestamptz NOT NULL, document_id

        uuid NOT NULL, seq smallint NOT NULL,
    ship_to_location_id uuid NOT NULL, bill_to_entity_id uuid NOT NULL, order_ids uuid[] NOT NULL,
    status text NOT NULL DEFAULT 'PLANNED' CHECK (status IN ('PLANNED','DELIVERED','UNDELIVERED','RECEIVED')),
    arrived_at timestamptz, pod_kind text CHECK (pod_kind IN ('PAPER','APP')), pod_signature_key text,

           pod_photo_key text,
    delivered_by uuid, undelivered_reason text, grn_document_id uuid, owner_entity_id uuid NOT NULL,
    PRIMARY KEY (received_at, drop_id), UNIQUE (received_at, document_id, seq)) PARTITION BY RANGE

           (received_at);
-- DN lines: kernel document_line with drop_id and batch_id in the extension below
CREATE TABLE trading.doc_delivery_line (line_id uuid NOT NULL, received_at timestamptz NOT NULL, drop_id uuid

        NOT NULL,
    order_line_id uuid, dispatched_qty numeric(14,3) NOT NULL, owner_entity_id uuid NOT NULL, PRIMARY KEY

           (received_at, line_id)) PARTITION BY RANGE (received_at);

CREATE TABLE trading.doc_grn (document_id uuid NOT NULL, received_at timestamptz NOT NULL, receiver_entity_id
        uuid NOT NULL,

    receiver_location_id uuid NOT NULL, seller_entity_id uuid, drop_id uuid, delivery_document_id uuid,
           purchase_order_ref text,

    supplier_id uuid, received_on date NOT NULL, counted_by uuid, confirmed_by uuid, confirmed_at timestamptz,
           reversal_of uuid,

    owner_entity_id uuid NOT NULL, PRIMARY KEY (received_at, document_id)) PARTITION BY RANGE (received_at);
CREATE TABLE trading.doc_grn_line (line_id uuid NOT NULL, received_at timestamptz NOT NULL, expected_qty

        numeric(14,3),
    received_qty numeric(14,3) NOT NULL CHECK (received_qty >= 0), damaged_qty numeric(14,3) NOT NULL DEFAULT 0

           CHECK (damaged_qty <= received_qty),
    batch_no varchar(40), manufacture_date date, expiry_date date, printed_mrp numeric(14,2), unit_cost

           numeric(14,4),
    owner_entity_id uuid NOT NULL, PRIMARY KEY (received_at, line_id)) PARTITION BY RANGE (received_at);

CREATE TABLE trading.doc_discrepancy (document_id uuid NOT NULL, received_at timestamptz NOT NULL,

           ('DAMAGED','EXPIRED_ON_ARRIVAL','WRONG_GOODS','QUALITY')),

CREATE TABLE trading.doc_payment_receipt (document_id uuid NOT NULL, received_at timestamptz NOT NULL,
       relationship_id uuid,

   payer_kind text NOT NULL CHECK (payer_kind IN ('ENTITY','CUSTOMER')), payer_entity_id uuid,
          payer_customer_account_id uuid,

   method text NOT NULL CHECK (method IN ('CASH','CHEQUE','TRANSFER','DEPOSIT')), reference text, received_on
          date NOT NULL,

   amount numeric(14,2) NOT NULL CHECK (amount > 0), unapplied_amount numeric(14,2) NOT NULL, owner_entity_id
          uuid NOT NULL,

   PRIMARY KEY (received_at, document_id)) PARTITION BY RANGE (received_at);
CREATE TABLE trading.cheque (receipt_document_id uuid PRIMARY KEY, bank text NOT NULL, cheque_no varchar(20)

       NOT NULL, dated date NOT NULL,
   status text NOT NULL DEFAULT 'RECEIVED' CHECK (status IN ('RECEIVED','CLEARED','BOUNCED')), outcome_at

          timestamptz, owner_entity_id uuid NOT NULL);

CREATE TABLE trading.relationship_exposure (relationship_id uuid PRIMARY KEY, seller_entity_id uuid NOT NULL,
       amount numeric(14,2) NOT NULL,

   open_invoices numeric(14,2) NOT NULL, accepted_not_invoiced numeric(14,2) NOT NULL, unapplied_credits
          numeric(14,2) NOT NULL,

   unapplied_receipts numeric(14,2) NOT NULL, as_of timestamptz NOT NULL, freshness timestamptz NOT NULL,
          warned_threshold smallint,

   owner_entity_id uuid NOT NULL);

CREATE TABLE trading.posting_map (doc_type_code varchar(8) NOT NULL, line_kind text NOT NULL, side text NOT
       NULL CHECK (side IN ('SELLER','BUYER')),

   debit_role text NOT NULL, credit_role text NOT NULL, amount_source text NOT NULL, PRIMARY KEY
          (doc_type_code, line_kind, side));

CREATE TABLE trading.transfer_request (request_id uuid PRIMARY KEY, mpcs_entity_id uuid NOT NULL,
       from_location_id uuid NOT NULL,

   to_location_id uuid NOT NULL, lines jsonb NOT NULL, status text NOT NULL DEFAULT 'REQUESTED' CHECK (status
          IN ('REQUESTED','APPROVED','REJECTED')),

   decided_by uuid, decided_at timestamptz, reason text, xfr_document_id uuid, owner_entity_id uuid NOT NULL);

-- PARTY masking view (19A 1): what the counterparty sees of an invoice
CREATE VIEW trading.v_invoice_party AS

   SELECT d.document_id, d.doc_number_display, d.owner_entity_id AS seller, d.counterparty_entity_id AS buyer,
          d.business_date,
                d.net_amount, d.tax_amount, d.gross_amount, i.due_date, i.settled_amount, i.credited_amount,
                       i.e_invoice_status

   FROM kernel.document d JOIN trading.doc_invoice i USING (document_id)
   WHERE d.doc_type_code = 'INV' AND (d.owner_entity_id = kernel.scope_entity() OR d.counterparty_entity_id =

          kernel.scope_entity()
              OR kernel.scope_class() = 'FEDERATION_VIEW');

-- unit_cost and internal notes are never in the view; lines are exposed through v_invoice_line_party (sku,
       qty, unit_price, tax, total)

6.1 ConfirmGrnHandler (the pivot) and the ingestion hook

@CommandHandler(permission = "shop.grn.confirm | whs.grn.confirm") // resolved by location type

@Transactional

GrnId handle(ConfirmGrn cmd, ScopeContext ctx) {

   Grn g = repo.draft(cmd.grnId()); require(g.receiverLocation in ctx.scope, "scope.mismatch");

   for (line : g.lines()) {                                                        // validation

      Sku sku = catalogue.getSku(line.skuId());

      require(line.receivedQty >= 0 && line.damagedQty <= line.receivedQty, "grn.line.quantities");

      if (sku.hasPrintedMrp && line.receivedQty > 0) require(line.printedMrp != null, "grn.line.mrp_required");

      if (sku.expiryTracked && line.receivedQty > 0) require(line.expiryDate != null,

      "grn.line.expiry_required");

      require(catalogue.conversionExists(sku, line.uom, today), "grn.line.uom");

   }

   // cost basis per line: trade price of the relationship for DN receipts (M3, tier by ordered qty), keyed

      unit_cost for local supply

   g.lines().forEach(l -> l.unitCost = g.isLocalSupply() ? l.unitCost :

      pricing.resolveTradePrice(g.relationship(), l.skuId(), l.uom, l.orderedQty(), today).price());

   // 1. issue: LOCATION series when receiverLocation is a SHOP, ENTITY series otherwise (kernel picks by

      document_type + location)

   documents.issue(g, ctx);                                                        // number, hash, state

      DRAFT->CONFIRMED, audit GRN_CONFIRMED

   // 2. batches (M2) - existing-or-create; synthetic when not batch-tracked

   for (line : g.lines()) if (line.receivedQty > 0)

      line.batchId = catalogue.registerBatch(new RegisterBatch(line.skuId, g.supplierId(), line.batchNo,

      line.manufactureDate, line.expiryDate, line.printedMrp, g.id()));

   // 3. the pivot event: M5 creates lots (qty received - damaged as GOOD, damaged as DAMAGED lot) and posts

      RECEIPT movements

   events.publish(GrnConfirmed.of(g));                                             // receiver, location, seller,

      drop, lines(batch, qty, damaged, unitCost), confirmedAt

   // 4. discrepancy

   if (DiscrepancyDetector.hasVariance(g)) { Discrepancy d = disputes.raise(g, ctx);

      events.publish(DiscrepancyRaised.of(d)); g.markDiscrepancy(); }

   else g.close();

   if (g.dropId() != null) drops.markReceived(g.dropId(), g.id());                 // DN closes when all drops

      terminal

   return g.id();

}
