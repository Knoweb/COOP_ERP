data, package layout, the API slice, handler specifications, the shared engine in Kotlin

consumers, the snapshot contributor, golden-basket fixtures, screens, tests and tickets.

wrap the tables and screens around it. The legal property ­ no output ever exceeds a control price ­ is a
property-based test, not a code review item.

// model

data class RetailLine(val skuId: Id, val uom: String, val price: Money, val tierFromQty: Qty = Qty.ZERO)

data class BatchCandidate(val batchId: Id, val batchNo: String, val printedMrp: Money?, val expiry:

LocalDate?, val onHand: Qty)

data class Ceiling(val price: Money, val uom: String, val ref: String)                       // control price,

unit-converted by the index

data class Policy(val kind: PolicyKind, val gapAmount: Money?, val gapPercent: BigDecimal?)

data class LineInput(val skuId: Id, val uom: String, val qty: Qty, val scannedBatchId: Id?, val pickedBatchId:

Id?)

enum class CapReason { NONE, MRP_LOWEST, MRP_BARCODE, MRP_PICKED, CONTROL_PRICE }

object PriceResolver {

fun resolveLine(i: LineInput, ix: PricingSnapshotIndex, date: LocalDate): LineResult {

val list = ix.retailLine(i.skuId, i.uom, date) ?: return LineResult.notSellable(i, "price.no_list_line")

val cands = ix.inStockBatches(i.skuId).filter { it.onHand > Qty.ZERO }

val policy = ix.policyFor(i.skuId)

val (batchTerm, batch, reasonIfBound) = when {

   !ix.sku(i.skuId).hasPrintedMrp -> Triple(null, i.scannedBatchId?.let(ix::batch), CapReason.NONE)

   policy.kind == BARCODE_RESOLVED && i.scannedBatchId != null ->

          ix.batch(i.scannedBatchId).let { Triple(it.printedMrp, it, CapReason.MRP_BARCODE) }

   policy.kind == PICKER && i.pickedBatchId != null ->

          ix.batch(i.pickedBatchId).let { Triple(it.printedMrp, it, CapReason.MRP_PICKED) }

   policy.kind == PICKER && gapExceeds(cands, policy) && i.pickedBatchId == null ->

          return LineResult.needsPick(i, cands)                                      // till shows the picker

   else -> cands.minByOrNull { it.printedMrp ?: Money.MAX }.let { Triple(it?.printedMrp, it,

          CapReason.MRP_LOWEST) }

}

val control = ix.ceilingFor(i.skuId, i.uom, date)                                    // lowest of sku- and

        tag-scoped

var unit = list.price; var reason = CapReason.NONE

if (batchTerm != null && batchTerm < unit) { unit = batchTerm; reason = reasonIfBound }

if (control != null && control.price < unit) { unit = control.price; reason = CapReason.CONTROL_PRICE }

return LineResult.base(i, batch, unit, mrpApplied = batchTerm, controlApplied = control?.price, capReason

        = reason)

object RuleEvaluator {

   fun applyLineRules(r: LineResult, ix: PricingSnapshotIndex, date: LocalDate, policy: StackingPolicy):

      LineResult {

      val applicable = ix.lineRulesFor(r.skuId, date).filter { it.matches(r) }     // kind-specific

             predicate

      val chosen = when (policy) {

      PRIORITY_THEN_BEST -> applicable.sortedWith(compareBy({ it.priority }, { -benefitOf(it, r)

             })).firstOrNull()

      BEST_ONLY             -> applicable.maxByOrNull { benefitOf(it, r) } }

      return chosen?.let { r.withBenefit(it.id, benefitOf(it, r), it.kind == EXPIRY_MARKDOWN) } ?: r

   }

   fun applyBillRule(lines: List<LineResult>, ix: PricingSnapshotIndex, date: LocalDate): BillBenefit? = ...

}

object BasketResolver {

   fun resolve(b: Basket, ix: PricingSnapshotIndex, date: LocalDate, tender: TenderKind): BasketResult {

      val lines = b.lines.map { PriceResolver.resolveLine(it, ix, date) }

      .map { if (it.sellable) RuleEvaluator.applyLineRules(it, ix, date, ix.stackingPolicy) else it }

      .map { it.recheckCeiling(ix, date) }                                         // step 6: never

             above control

      val bill = RuleEvaluator.applyBillRule(lines, ix, date)

      val taxed = Tax.decompose(lines, bill, ix)                                   // inclusive -> tax

             amounts, apportion bill benefit

      val total = taxed.sumOf { it.lineTotal } - (bill?.amount ?: Money.ZERO)

      val rounding = if (tender == CASH) Rounding.toRupee(total) else Money.ZERO

      return BasketResult(taxed, bill, total + rounding, rounding, ENGINE_VERSION, ix.snapshotVersion)

   }

}

Aspect                  Design
Index
                        PricingSnapshotIndex is built once per snapshot version from the snapshot rows: maps sku  retail lines by
Determinism             uom and tier, sku/tag  active rules, sku/tag  ceilings (unit-converted using M2 conversions), sku  policy, sku
                         in-stock batches (from the lot rows). Building it is O(rows); resolving is O(lines × rules-for-sku).
Central use
                        No clocks, no randomness, no I/O; the business date is an input; money arithmetic uses the library's fixed
                        rounding modes; the same inputs give the same bytes on JVM and Android (a golden fixture asserts the
                        serialised BasketResult).

                        PricingQueriesImpl.resolveBasket loads the same inputs from tables into the same index type and calls
                        the same function; ingestion re-validation rebuilds the index for the snapshot version the till reported (kernel
                        keeps snapshot versions for the change-log window).
