Read this first. Build the ledger core first: LedgerService.post(movements) is the one place stock changes, and
everything else ­ receipts, sales, counts, write-offs, repacks, transfers, opening balances ­ is a caller that assembles
movements and cites a document. Get the property test 'lot balance equals movement sum under any interleaving'
green on day two, then never touch qty_on_hand from anywhere else.

6. The ledger core and handler specifications

6.1 LedgerService and CostService

// ledger/LedgerService.post(List<Movement> ms, Document doc, ScopeContext ctx) -- the only writer of

      stock_lot.qty_on_hand

@Transactional

void post(ms, doc, ctx) {

   for (m : ms) {

      Lot lot = lots.lockOrCreate(m.locationId, m.batchId, m.condition, defaultCost = m.unitCost ?:

      cost.avg(ctx.entityId, skuOf(m.batchId))) // SELECT ... FOR UPDATE

      long seq = sequences.next(m.locationId, source = ctx.isDevice ? m.deviceSeqSource : "central")

      // dense per (location, source)

      movements.insert(m.withSeq(seq).withCost(costFor(m, lot, ctx)).citing(doc))

      lot.qtyOnHand += m.qtyDelta; lot.lastMovementSeq = seq

      if (lot.qtyOnHand < 0 && lot.negativeSince == null) { lot.negativeSince = now;

      events.publish(LotNegative.of(lot, doc)); audit REVIEW NEGATIVE_STOCK }

      if (lot.qtyOnHand >= 0 && lot.negativeSince != null) { lot.negativeSince = null; audit INFO

      NEGATIVE_CLEARED }

      lots.save(lot)

      cost.apply(m, ctx.entityId)

      // below

      events.publish(StockMoved.of(m, lot, doc))

   }

}

costFor(m, lot, ctx) = switch (m.type) { RECEIPT, OPENING_BALANCE, REPACK_PRODUCE -> m.unitCost; TRANSFER_IN

      -> m.unitCost (source lot cost);

                                             default (issues: SALE, WRITE_OFF, REPACK_CONSUME, COUNT_ADJUST-,

                                              TRANSFER_OUT, RETURN) -> cost.avg(entity, sku) }

// ledger/CostService.apply(m, entity): moving weighted average per (entity, sku) -- doc 25 3.3

row = costs.lock(entity, sku)

if (m.isIntake() && m.type != TRANSFER_IN) { row.avg = (row.qty * row.avg + m.qty * m.unitCost) / (row.qty +

      m.qty); row.qty += m.qty }

else if (m.type == TRANSFER_IN) { row.qty += m.qty }                               // within the entity: no cost change

else { row.qty += m.qtyDelta }                                                     // issues change qty, not avg

// avg never negative; if row.qty <= 0 after an issue, avg is kept (last known) - property test asserts
