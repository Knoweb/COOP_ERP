5. Boundary enforcement in the build

Rules R1­R7 of doc 17 §6.1 become tests that run on every commit. They are the only reason a modular
monolith stays modular.

// backend/app/src/test/java/lk/coopfed/knoweb/ArchitectureTests.java
class ArchitectureTests {

   static final ApplicationModules modules = ApplicationModules.of(KnowebApplication.class);

   @Test void modulithStructureIsValid() {           // R1, R5: published packages, allowed dependencies
      modules.verify();

   }

   @Test void documentsTheModuleGraph() {            // keeps docs/modules/ current
      new Documenter(modules).writeDocumentation();

   }

   static final JavaClasses classes = new ClassFileImporter()
          .withImportOption(DO_NOT_INCLUDE_TESTS).importPackages("lk.coopfed.knoweb");

   @Test void layersAreRespected() {                 // R2, R3

      layeredArchitecture().consideringOnlyDependenciesInLayers()

      .layer("kernel").definedBy("..kernel..")

      .layer("master").definedBy("..m1party..", "..m2catalogue..", "..m3pricing..")

      .layer("transactions").definedBy("..m4trading..", "..m5inventory..", "..m6pos..",

                                         "..m7customers..", "..m10procurement..")

      .layer("read").definedBy("..m8reporting..", "..m9integration..")

      .whereLayer("master").mayOnlyAccessLayers("kernel")

      .whereLayer("transactions").mayOnlyAccessLayers("kernel", "master")

      .whereLayer("read").mayOnlyAccessLayers("kernel", "master", "transactions")

      .check(classes);

   }

   @Test void noCrossSchemaRepositories() {          // R4: a repository touches its own schema only

      classes().that().areAnnotatedWith(Repository.class)

      .should(new OwnSchemaOnly()).check(classes); // custom condition: @Table(schema=) matches module

   }

   @Test void commandsCarryPermissions() {           // R7: no handler without a permission

      classes().that().areAnnotatedWith(CommandHandler.class)

      .should(new HavePermissionAttribute()).check(classes);

   }

   @Test void kernelImportsNoModule() {              // R5

      noClasses().that().resideInAPackage("..kernel..")

      .should().dependOnClassesThat().resideInAnyPackage("..m1party..", "..m2catalogue..", /* ... */

      "..hello..")

      .check(classes);

   }

}

Two further checks live in the pipeline rather than in Java: a schema-ownership script that parses each module's
migrations and fails if a migration touches another module's schema, and an i18n check that fails the build when
a message id lacks any of the three languages (restated from doc 19 §5.2).

-- session scope variables read by RLS policies (doc 18 3.7); set by the kernel scope filter
-- app.scope_entity_id, app.scope_location_id, app.scope_class (custom GUCs, no DDL needed)

-- helper used by every RLS policy
CREATE OR REPLACE FUNCTION kernel.scope_entity() RETURNS uuid

   LANGUAGE sql STABLE AS $$ SELECT nullif(current_setting('app.scope_entity_id', true), '')::uuid $$;
CREATE OR REPLACE FUNCTION kernel.scope_location() RETURNS uuid

   LANGUAGE sql STABLE AS $$ SELECT nullif(current_setting('app.scope_location_id', true), '')::uuid $$;
CREATE OR REPLACE FUNCTION kernel.scope_class() RETURNS text

   LANGUAGE sql STABLE AS $$ SELECT coalesce(current_setting('app.scope_class', true), 'NONE') $$;

                      db/migration//V__.sql; numbering is per module location (Flyway keeps one history table per schema

Partitioning          A migration touches only its module's schema; references to other schemas are by uuid columns without foreign

6.3 Row-level security policy template

-- applied to every operational table; classes from doc 18 3.7
ALTER TABLE party.location ENABLE ROW LEVEL SECURITY;
ALTER TABLE party.location FORCE ROW LEVEL SECURITY;

CREATE POLICY own_read ON party.location FOR SELECT TO app_rw
    USING (owner_entity_id = kernel.scope_entity()
                 AND (kernel.scope_location() IS NULL OR location_id = kernel.scope_location()));

CREATE POLICY own_write ON party.location FOR INSERT TO app_rw
    WITH CHECK (owner_entity_id = kernel.scope_entity());

CREATE POLICY fed_view ON party.location FOR SELECT TO app_rw
    USING (kernel.scope_class() = 'FEDERATION_VIEW');

CREATE POLICY ext_view ON party.location FOR SELECT TO app_rw
    USING (kernel.scope_class() = 'EXTERNAL_TIMEBOXED'
                 AND owner_entity_id = ANY (kernel.granted_entities())); -- function provided by 19A

-- PARTY policies exist only on document tables and use party columns (seller/buyer) - see 19A

class RegisterGreetingHandler implements Handles<RegisterGreeting, GreetingId> {

   @Transactional
   public GreetingId handle(RegisterGreeting cmd, ScopeContext scope) {

       // 1. guards (business rules) - throw ProblemException("hello.greeting.duplicate") etc.
       // 2. mutation
       Greeting g = Greeting.create(Ids.next(), scope.entityId(), cmd.textEn(), cmd.textSi(), cmd.textTa());
       repo.save(g);
       // 3. audit in the same transaction
       audit.record(AuditType.HELLO_GREETING_REGISTERED, Subject.of(g), null, g.snapshot(), scope);
       // 4. event in the same transaction (outbox)
       events.publish(new GreetingRegistered(g.id(), scope.entityId()));
       return g.id();
   }
}
