the analyst role setup, dashboards, screens, tests and tickets. Self-contained.

Read this first. Every table in the reporting schema must be droppable: build the projection consumers so that
RebuildProjection(name, 0) produces the same rows as the live consumer did, and make the equivalence test
the first test you write. If a projection cannot be rebuilt from the archive, it is not a projection; it is a second source of
truth, and it does not belong here.

6. The projection pattern, freshness and end-of-day

// projection/Projection.kt - every consumer is an instance of this pattern

abstract class Projection<E : DomainEvent>(val name: String, val types: Set<String>) {

   abstract fun upsert(e: E, tx: Tx)                        // idempotent by aggregate id (+ event id for facts)

   @EventConsumer(types = types, consumer = "m8.$name")     // kernel inbox guarantees once-per-event

   fun on(e: E) = tx { upsert(e, it); state.advance(name, e.sourceSeq, e.receivedAt) }

   fun rebuild(fromSeq: Long) {                             // RebuildService

      state.mark(name, REBUILDING, shadow = "${name}_shadow"); createShadowLike(name)

      archive.replay(types, fromSeq) { e -> tx { upsertInto(shadow, e, it) } } // reads kernel.event_outbox

               partitions in (source, seq) order

      tx { swap(name, shadow); state.mark(name, LIVE, lastSeq = archive.headSeq()) } // ALTER TABLE ... RENAME

               in one transaction; consumer resumes

   }

}

// receipt_line_fact.upsert: INSERT ... ON CONFLICT (received_at, line_id) DO NOTHING; voided -> UPDATE

      receipt_fact.voided_by (fact rows are never deleted)

// stock_position.upsert: INSERT ... ON CONFLICT (location, batch, condition) DO UPDATE SET qty_on_hand =

      qty_on_hand + delta, ... (delta applied once thanks to the inbox)

// canonical_sku_id = catalogue.resolveAlias(e.skuId) at projection time; a later merge event re-points

      historical facts by a corrective job (sku.merged.v1 consumer)

// freshness/FreshnessConsumer: on sync.batch_received.v1 -> location_freshness.applied_through =
       max(existing, batch.lastEventReceivedAt); devices_reporting

// FreshnessResolver.forLocations(ids): { min(applied_through), count(applied_through < now - staleThreshold)
       }

// sales_daily.freshness = FreshnessResolver.forLocations([location]).min at the time of the last upsert
       (re-stamped by LateEventHandler)

// eod/EndOfDayConsumer: on location.day_closed.v1(location, businessDate) ->
// recompute sales_daily and cash_daily rows for (location, businessDate) from facts (cheap: one location,

       one day); stamp freshness;
// publish end_of_day.completed.v1 with totals. LateEventHandler: any fact for a closed (location, day)

       triggers the same recompute.
// RollupJob (monthly): sales_monthly by entity/district/tag from sales_daily; FY roll-ups by

       entity.financial_year_start_month (M1)
